uintptr_t GetModuleBaseAddress(DWORD procId, const wchar_t* modName) {
    uintptr_t base = 0;
    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPMODULE | TH32CS_SNAPMODULE32, procId);
    if (snapshot != INVALID_HANDLE_VALUE) {
        MODULEENTRY32W modEntry;
        modEntry.dwSize = sizeof(modEntry);
        if (Module32FirstW(snapshot, &modEntry)) {
            do {
                if (_wcsicmp(modEntry.szModule, modName) == 0) {
                    base = (uintptr_t)modEntry.modBaseAddr;
                    break;
                }
            } while (Module32NextW(snapshot, &modEntry));
        }
        CloseHandle(snapshot);
    }
    return base;
}

DWORD GetProcessIdByName(const wchar_t* name) {
    DWORD pid = 0;
    HANDLE snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (snapshot != INVALID_HANDLE_VALUE) {
        PROCESSENTRY32W procEntry;
        procEntry.dwSize = sizeof(procEntry);
        if (Process32FirstW(snapshot, &procEntry)) {
            do {
                if (_wcsicmp(procEntry.szExeFile, name) == 0) {
                    pid = procEntry.th32ProcessID;
                    break;
                }
            } while (Process32NextW(snapshot, &procEntry));
        }
        CloseHandle(snapshot);
    }
    return pid;
}

template<typename T>
bool ReadMemory(uintptr_t address, T& out) {
    return ReadProcessMemory(g_hProcess, (LPCVOID)address, &out, sizeof(T), nullptr) != 0;
}

bool ReadBytes(uintptr_t address, void* buffer, size_t size) {
    return ReadProcessMemory(g_hProcess, (LPCVOID)address, buffer, size, nullptr) != 0;
}

Vector3 ReadVector3(uintptr_t address) {
    Vector3 v{};
    ReadMemory(address, v);
    return v;
}

std::string ReadString(uintptr_t address, size_t maxLen = 64) {
    char buffer[256] = {0};
    if (ReadBytes(address, buffer, maxLen)) {
        return std::string(buffer);
    }
    return "";
}

bool WorldToScreen(const Vector3& worldPos, const ViewMatrix& vm, int screenW, int screenH, Vector2& screenPos) {
    float w = vm.m[3][0] * worldPos.x + vm.m[3][1] * worldPos.y + vm.m[3][2] * worldPos.z + vm.m[3][3];
    if (w < 0.001f) return false;
    float x = vm.m[0][0] * worldPos.x + vm.m[0][1] * worldPos.y + vm.m[0][2] * worldPos.z + vm.m[0][3];
    float y = vm.m[1][0] * worldPos.x + vm.m[1][1] * worldPos.y + vm.m[1][2] * worldPos.z + vm.m[1][3];
    screenPos.x = (screenW / 2.0f) * (1.0f + x / w);
    screenPos.y = (screenH / 2.0f) * (1.0f - y / w);
    return true;
}

// ============ OVERLAY WINDOW ============
LRESULT CALLBACK OverlayWndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
    switch (msg) {
    case WM_DESTROY:
        PostQuitMessage(0);
        return 0;
    case WM_PAINT:
        ValidateRect(hwnd, nullptr);
        return 0;
    }
    return DefWindowProcW(hwnd, msg, wParam, lParam);
}

bool CreateOverlayWindow(HINSTANCE hInstance) {
    WNDCLASSEXW wc = {};
    wc.cbSize = sizeof(WNDCLASSEXW);
    wc.lpfnWndProc = OverlayWndProc;
    wc.hInstance = hInstance;
    wc.lpszClassName = L"RustOverlayClass";
    wc.hCursor = LoadCursor(nullptr, IDC_ARROW);
    if (!RegisterClassExW(&wc)) return false;

    int screenW = GetSystemMetrics(SM_CXSCREEN);
    int screenH = GetSystemMetrics(SM_CYSCREEN);

    g_overlayWnd = CreateWindowExW(
        WS_EX_TOPMOST | WS_EX_TRANSPARENT | WS_EX_LAYERED | WS_EX_TOOLWINDOW,
        wc.lpszClassName,
        L"Overlay",
        WS_POPUP,
        0, 0, screenW, screenH,
        nullptr, nullptr, hInstance, nullptr);

    if (!g_overlayWnd) return false;

    SetLayeredWindowAttributes(g_overlayWnd, RGB(0, 0, 0), 0, LWA_COLORKEY);
    MARGINS margins = { -1, -1, -1, -1 };
    DwmExtendFrameIntoClientArea(g_overlayWnd, &margins);

    ShowWindow(g_overlayWnd, SW_SHOW);
    UpdateWindow(g_overlayWnd);
    return true;
}

bool InitD2D() {
    if (FAILED(D2D1CreateFactory(D2D1_FACTORY_TYPE_SINGLE_THREADED, &g_d2dFactory)))
        return false;

    RECT rc;
    GetClientRect(g_overlayWnd, &rc);
    D2D1_SIZE_U size = D2D1::SizeU(rc.right - rc.left, rc.bottom - rc.top);

    if (FAILED(g_d2dFactory->CreateHwndRenderTarget(
        D2D1::RenderTargetProperties(D2D1_RENDER_TARGET_TYPE_DEFAULT,
            D2D1::PixelFormat(DXGI_FORMAT_B8G8R8A8_UNORM, D2D1_ALPHA_MODE_PREMULTIPLIED)),
        D2D1::HwndRenderTargetProperties(g_overlayWnd, size),
        &g_renderTarget)))
        return false;

    if (FAILED(DWriteCreateFactory(DWRITE_FACTORY_TYPE_SHARED, __uuidof(IDWriteFactory),
        reinterpret_cast<IUnknown**>(&g_dwriteFactory))))
        return false;

    if (FAILED(g_dwriteFactory->CreateTextFormat(
        L"Arial", nullptr,
        DWRITE_FONT_WEIGHT_NORMAL,
        DWRITE_FONT_STYLE_NORMAL,
        DWRITE_FONT_STRETCH_NORMAL,
        12.0f, L"en-us", &g_textFormat)))
        return false;

    g_renderTarget->CreateSolidColorBrush(D2D1::ColorF(D2D1::ColorF::Red, 1.0f), &g_brush);
    return true;
}
