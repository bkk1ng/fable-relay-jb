            if (!_wcsicmp(pe.szExeFile, name)) {

                break;
            }
        } while (Process32NextW(snap, &pe));
    }
    CloseHandle(snap);
    return pid;
}

// Convert bone-space coordinates to world-space using ComponentToWorld transform
Vector3 TransformPoint(const FTransform& transform, const Vector3& point) {
    // Apply scale
    Vector3 scaled(point.x * transform.Scale[0],
                   point.y * transform.Scale[1],
                   point.z * transform.Scale[2]);
    
    // Apply rotation (quaternion)
    float qx = transform.Rotation[0];
    float qy = transform.Rotation[1];
    float qz = transform.Rotation[2];
    float qw = transform.Rotation[3];
    
    // Quaternion rotation
    Vector3 u(qx, qy, qz);
    float s = qw;
    
    // v' = 2 * (u · v) * u + (s^2 - u·u) * v + 2 * s * (u × v)
    float dot_uv = 2.0f * u.dot(scaled);
    float s2_minus_u2 = s*s - u.dot(u);
    Vector3 cross(2.0f * s * (u.y * scaled.z - u.z * scaled.y),
                  2.0f * s * (u.z * scaled.x - u.x * scaled.z),
                  2.0f * s * (u.x * scaled.y - u.y * scaled.x));
    
    Vector3 rotated;
    rotated.x = dot_uv * u.x + s2_minus_u2 * scaled.x + cross.x;
    rotated.y = dot_uv * u.y + s2_minus_u2 * scaled.y + cross.y;
    rotated.z = dot_uv * u.z + s2_minus_u2 * scaled.z + cross.z;
    
    // Apply translation
    return Vector3(rotated.x + transform.Translation[0],
                   rotated.y + transform.Translation[1],
                   rotated.z + transform.Translation[2]);
}

    *screenY = sy;

}
